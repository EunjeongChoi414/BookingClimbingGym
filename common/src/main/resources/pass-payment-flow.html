의사코드

CLIENT — Pass 구매 흐름
purchasePass(gymId, passId):

  // 1. 서버에 결제 준비 요청
  order = POST /app/gyms/{gymId}/passes/{passId}/payment/prepare
  // ← { orderId, amount, passName }

  // 2. 토스페이먼츠 SDK 초기화
  toss = TossPayments(CLIENT_KEY)
  payment = toss.payment("카드")

  try:
    // 3. 결제 위젯 실행 (사용자 카드 입력)
    result = await payment.requestPayment({
      orderId   : order.orderId,
      orderName : order.passName,
      amount    : order.amount,
      successUrl: "{baseUrl}/payment/success",
      failUrl   : "{baseUrl}/payment/fail"
    })

    // 4. 임시 승인 콜백 → 서버 confirm 요청 (여기서 결제 확정)
    userPass = POST /app/gyms/{gymId}/passes/{passId}/payment/confirm
               body: result

    // 5. Pass 발급 완료 UI 표시
    showSuccess(userPass)

  catch PaymentCancelledError:
    showMessage("결제가 취소되었습니다.")

  catch PaymentFailedError as e:
    showMessage(e.message)
SERVER — POST /payment/prepare
preparePayment(token, gymId, passId):

  // 1. 유저 인증
  userId = getUserIdFromToken(token)

  // 2. 암장 & Pass 존재 확인
  gym  = gymRepository.getById(gymId)
        → GymNotFoundException if 없음

  pass = gym.getPassById(passId)
        → PassNotFoundException if 없음

  // 3. 중복 PENDING 주문 방지
  if orderRepository.existsPending(userId, passId):
    throw DuplicatePendingOrderException

  // 4. 주문 생성 및 저장
  orderId = UUID.randomUUID()
  order = new Order(
    orderId  : orderId,
    userId   : userId,
    passId   : passId,
    amount   : pass.getPrice(),
    status   : PENDING,
    expiresAt: now + 10분
  )
  orderRepository.save(order)

  return { orderId, amount: pass.getPrice(), passName: pass.getName() }
SERVER — POST /payment/confirm
confirmPayment(token, gymId, passId,
               paymentKey, orderId, amount):

  userId = getUserIdFromToken(token)

  // 1. 위변조 방지: 서버 저장 주문과 비교
  order = orderRepository.getByOrderId(orderId)
        → OrderNotFoundException if 없음

  if order.userId ≠ userId:
    throw UnauthorizedException
  if order.amount ≠ amount:
    throw AmountMismatchException
  if order.status ≠ PENDING:
    throw AlreadyProcessedException
  if order.expiresAt < now():
    throw OrderExpiredException

  // 2. 토스페이먼츠 결제 승인 API 호출
  tossResponse = callTossConfirmAPI(paymentKey, orderId, amount)

  if tossResponse.isFailure():
    order.status = FAILED
    orderRepository.save(order)
    throw PaymentFailedException(tossResponse.errorMessage)

  // 3. 결제 완료 처리
  order.status     = COMPLETED
  order.paymentKey = paymentKey
  order.receiptUrl = tossResponse.receipt.url
  orderRepository.save(order)

  // 4. UserPass 발급
  pass    = gym.getPassById(passId)
  user    = userRepository.getById(userId)
  userPass = new UserPass(pass, user, validFrom=today())
  userPassRepository.save(userPass)

  return {
    userPassId    : userPass.getId(),
    passName      : pass.getName(),
    validUntil    : userPass.getValidUntil(),
    remainingUses : userPass.getRemainingUses()
  }
TOSS — 결제 승인 API 호출
callTossConfirmAPI(paymentKey, orderId, amount):

  secretKey = Base64("{TOSS_SECRET_KEY}:")

  response = HTTP_POST(
    url    : "https://api.tosspayments.com/v1/payments/confirm",
    headers: {
      Authorization : "Basic {secretKey}",
      Content-Type  : "application/json"
    },
    body   : {
      paymentKey: paymentKey,
      orderId   : orderId,
      amount    : amount
    }
  )

  if response.status == 200:
    return TossPaymentResult(
      success   : true,
      receiptUrl: response.body.receipt.url,
      method    : response.body.method,
      approvedAt: response.body.approvedAt
    )

  else:  // 4xx / 5xx
    return TossPaymentResult(
      success     : false,
      errorCode   : response.body.code,
      errorMessage: response.body.message
    )

// 주요 토스 에러 코드
// ALREADY_PROCESSED_PAYMENT → 중복 결제 시도
// INVALID_STOPPED_CARD      → 정지 카드
// EXCEED_MAX_DAILY_PAYMENT_COUNT → 한도 초과

주요 에러 처리

AmountMismatchException

클라이언트가 보낸 amount와 서버 저장 amount 불일치 → 400 Bad Request (위변조 의심)

AlreadyProcessedException

이미 COMPLETED/FAILED 상태의 orderId로 confirm 재시도 → 409 Conflict

OrderExpiredException

10분이 지난 PENDING 주문 → 410 Gone, 클라이언트에서 재준비 유도

PaymentFailedException

토스 승인 거절 (카드 한도, 정지 등) → 402 Payment Required + 토스 에러 코드 전달

DuplicatePendingOrderException

같은 유저가 같은 Pass에 PENDING 주문이 이미 존재 → 409 Conflict (기존 orderId 반환)

PassNotFoundException

해당 암장에 존재하지 않는 passId → 404 Not Found
